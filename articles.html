incremental_pid.c 
float PID_Calc(PID_TypeDef *pid, float target, float current) {
    const float error = target - current;
    const float delta = pid->Kp * (error - pid->last_error)
                      + pid->Ki * error
                      + pid->Kd * (error - 2.0f * pid->last_error + pid->prev_error);

    pid->output += delta;
    pid->prev_error = pid->last_error;
    pid->last_error = error;
    return pid->output;
}
